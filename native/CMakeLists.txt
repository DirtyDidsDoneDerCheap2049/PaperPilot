cmake_minimum_required(VERSION 3.20)
project(reader_tasks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(READER_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../work/native-vendor" CACHE PATH "Downloaded pinned dependencies")
add_library(reader_sqlite STATIC "${READER_VENDOR}/sqlite3.c")
target_compile_definitions(reader_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_include_directories(reader_sqlite PUBLIC "${READER_VENDOR}")
add_executable(reader_tasks main.cpp)
target_link_libraries(reader_tasks PRIVATE reader_sqlite)
set(MYSQL_CLIENT_DIR "$ENV{MYSQL_CLIENT_DIR}" CACHE PATH "MySQL client installation")
if(MYSQL_CLIENT_DIR)
  target_compile_definitions(reader_tasks PRIVATE READER_MYSQL)
  target_include_directories(reader_tasks PRIVATE "${MYSQL_CLIENT_DIR}/include")
  find_library(MYSQL_CLIENT_LIBRARY NAMES libmysql mysqlclient PATHS "${MYSQL_CLIENT_DIR}/lib" REQUIRED NO_DEFAULT_PATH)
  target_link_libraries(reader_tasks PRIVATE "${MYSQL_CLIENT_LIBRARY}")
endif()
if(WIN32)
  target_link_libraries(reader_tasks PRIVATE shell32)
endif()
if(UNIX)
  target_link_libraries(reader_tasks PRIVATE pthread dl m)
endif()
install(TARGETS reader_tasks RUNTIME DESTINATION bin)
